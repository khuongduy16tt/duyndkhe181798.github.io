        function switchTab(tabId) {
            // Hide all tab content
            const contents = document.querySelectorAll('.tab-content');
            contents.forEach(content => content.classList.add('hidden'));
            contents.forEach(content => content.classList.remove('block'));

            // Show active tab
            const activeTab = document.getElementById(tabId);
            activeTab.classList.remove('hidden');
            activeTab.classList.add('block');

            // Reset navigation button classes
            const navButtons = document.querySelectorAll('.nav-btn');
            navButtons.forEach(btn => {
                btn.className = "nav-btn px-4 py-2 rounded-lg text-sm font-semibold transition duration-200 text-slate-600 hover:text-crimson-700 hover:bg-slate-50";
            });

            // Highlight current button
            const activeBtn = document.getElementById('btn-' + tabId);
            if (activeBtn) {
                activeBtn.className = "nav-btn px-4 py-2 rounded-lg text-sm font-semibold transition duration-200 text-crimson-700 bg-gold-100";
            }

            // Scroll to view
            window.scrollTo({ top: activeTab.offsetTop - 120, behavior: 'smooth' });
        }
